% ---------------- environments ----------------
% Open issue for discussion with the supervisors. Optional argument = title.
\newenvironment{openissue}[1][Open]{%
  \setbeamercolor{block title alerted}{fg=white,bg=issuecolor}%
  \setbeamercolor{block body alerted}{fg=black,bg=issuecolor!10}%
  \setbeamerfont{itemize/enumerate body}{size=\small}%
  \setbeamerfont{itemize/enumerate subbody}{size=\footnotesize}%
  \begin{alertblock}{\small #1}\small}{\end{alertblock}}

% Key result box. Optional argument = title.
\newenvironment{keyresult}[1][Key point]{%
  \setbeamercolor{block title example}{fg=white,bg=keycolor}%
  \setbeamercolor{block body example}{fg=black,bg=keycolor!8}%
  \begin{exampleblock}{\small #1}}{\end{exampleblock}}
